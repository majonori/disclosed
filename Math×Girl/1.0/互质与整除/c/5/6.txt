namespace Miller_Rabin{
	const int Pcnt=7;
	const int64_t p[Pcnt]={2,325,9375,28178,450775,9780504,1795265022};
	int64_t pow(int64_t a,int64_t b,int64_t p){
		int64_t ans=1;
		for(;b;a=(__int128)a*a%p,b>>=1)if(b&1)ans=(__int128)ans*a%p;
		return ans;
	}
	bool check(int64_t x,int64_t p){
		if(x%p==0||pow(p%x,x-1,x)^1)return 1;
		int64_t t,k=x-1;
		while((k^1)&1){
			t=pow(p%x,k>>=1,x);
			if(t^1&&t^(x-1))return 1;
			if(!(t^(x-1)))return 0;
		}return 0;
	}
	bool prime(int64_t x){
		if(x<2)return 0;
		for(int64_t i=0;i!=Pcnt;++i){
			if(x<=p[i])return 1;
			if(check(x,p[i]))return 0;
		}return 1;
	}
}using Miller_Rabin::prime;

const double PI2=6.2831853071795864769l;
const i32 MAXS=16,MAXN=104000,MAXA=64;
const i64 P=1000000007;

i64 T,s,p[MAXA],a[MAXA],ta[MAXA];

struct Vec{
	i32 t[MAXS];
  Vec(){memset(t,0,sizeof(i32)*MAXS);}
	bool dbl(Vec v){
		bool flag=1;
    for(i32 i=0;i!=s;i++)if(t[i]!=v[i])
      flag=0,t[i]=std::min(t[i]<<1|1,v[i]);
    return flag;
	}
  bool next(Vec v,i32 w=-1){
    for(i32 i=0;i!=s;i++){
      if(i==w||t[i]==v[i]){t[i]=0;continue;}
      t[i]++;return 1;
    }return 0;
  }
  i32&operator[](i32 x){return t[x];}
  bool operator+=(Vec v){
    for(i32 i=0;i!=s;i++)
      if(t[i]+=v[i]>a[i])return 0;
    return 1;
  }
  Vec operator-(Vec v){
    Vec x;
    for(i32 i=0;i!=s;i++)x[i]=t[i]-v[i];
    return x;
  }
};
i64 fpow(i64 a,i64 n=P-2){
  i64 x=1;
  for(;n;(a*=a)%=P,n>>=1)if(n&1)(x*=a)%=P;
  return x;
}
i64 M(Vec v){
  i64 x=0;
  for(i64 i=0,w=1;i!=s;w*=(a[i++]+1))x+=v[i]*w;
  return x;
}
i64 D(Vec v){
  i64 x=1;
  for(i64 i=0;i!=s;i++)x*=(v[i]+1);
  return x;
}
i64 V(Vec v){
  i64 x=1;
  for(i64 i=0;i!=s;i++)x=x*powl(p[i],v[i])+.5;
  return x;
}

void Init(){
  for(i32 i=1;i!=MAXA<<1;i++)Inv[i]=fpow(i);
  for(i32 i=0,j=1;i!=MAXA<<1;i++)
    for(nCr[i][0]=1;j<i;j++)
      nCr[i][j]=(nCr[i-1][j-1]+nCr[i-1][j])%P;
}

void DFT(C*f,i32 feet,i32 len,i32 flag){
  static C F[MAXA];
  for(i32 i=0;i<=len;i++)F[i]=f[i*feet];
  C unit={cos(PI2/(len+1)),flag*sin(PI2/(len+1))},w1={1,0};
  for(i32 i=0;i<=len;i++,w1*=unit){
    C w={1,0};f[i*feet]=0;
    for(i32 j=0;j<=len;j++,w*=w1)f[i*feet]+=F[j]*w;
    if(flag==-1)f[i*feet]/=len+1;
  }
}
void DFT(C*f,Vec v,int flag=1){
  for(i32 i=(flag==1?0:s-1);i!=(flag==1?s:-1);i+=flag){
    Vec d;do DFT(f+M(d),ta[i],v[i],flag);
    while(d.next(v,i));
  }
}

int *times(int*f,int*g,Vec n){
	static C F[MAXN];Vec d;
	do F[M(d)]={f[M(d)],g[M(d)]};while(d.next(n));
	DFT(F,n);
	do F[M(d)]*=F[M(d)];while(d.next(n));
	DFT(F,n,-1);
//	for(int i=0;i!=20;i++)std::cout<<F[i];std::cout<<"\n\n";
	do f[M(d)]=i64(F[M(d)].imag()/2+.5)%P,F[M(d)]=0;
	while(d.next(n));return f;
}

int *inv(int*f,Vec n){
	static int G[MAXN<<1],g[MAXN<<1];
	G[0]=fpow(f[0]);Vec d;
	while(d.dbl(n)){
		Vec i;do g[M(i)]=G[M(i)];while(i.next(d));
		times(times(g,G,d),f,d);
		do G[M(i)]=(G[M(i)]*2ll-g[M(i)]+P)%P;while(i.next(d));
	}do f[M(d)]=G[M(d)],G[M(d)]=g[M(d)]=0;while(d.next(n));
	return f;
}
